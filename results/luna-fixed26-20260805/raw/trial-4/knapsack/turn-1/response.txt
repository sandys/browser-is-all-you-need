knapsack.cpp
```
#include "knapsack.h"

#include <algorithm>
#include <vector>

namespace knapsack
{

int maximum_value(int maximum_weight, const std::vector<Item>& items)
{
    if (maximum_weight <= 0)
    {
        return 0;
    }

    std::vector<int> best_value(maximum_weight + 1, 0);

    for (const Item& item : items)
    {
        if (item.weight > maximum_weight)
        {
            continue;
        }

        for (int capacity = maximum_weight; capacity >= item.weight; --capacity)
        {
            best_value[capacity] = std::max(
                best_value[capacity],
                best_value[capacity - item.weight] + item.value);
        }
    }

    return best_value[maximum_weight];
}

} // namespace knapsack
```
