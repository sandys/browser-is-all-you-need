The compilation errors occur because the tests pass allergen names as string literals, while `is_allergic_to` only accepts `allergen`. Update the interface and implementation to accept `std::string`, while retaining the enum overload for compatibility. The returned allergy set contains the corresponding allergen names.

allergies.h
```
#if !defined(ALLERGIES_H)
#define ALLERGIES_H

#include <string>
#include <unordered_set>

namespace allergies {

enum class allergen : unsigned {
    eggs = 1,
    peanuts = 2,
    shellfish = 4,
    strawberries = 8,
    tomatoes = 16,
    chocolate = 32,
    pollen = 64,
    cats = 128
};

class allergy_test {
public:
    explicit allergy_test(unsigned score);

    bool is_allergic_to(const std::string& item) const;
    bool is_allergic_to(allergen item) const;

    std::unordered_set<std::string> get_allergies() const;

private:
    unsigned score_;
};

}  // namespace allergies

#endif  // ALLERGIES_H

allergies.cpp
```
#include "allergies.h"

namespace allergies {

namespace {

struct allergen_entry {
    const char* name;
    allergen item;
};

const allergen_entry known_allergens[] = {
    {"eggs", allergen::eggs},
    {"peanuts", allergen::peanuts},
    {"shellfish", allergen::shellfish},
    {"strawberries", allergen::strawberries},
    {"tomatoes", allergen::tomatoes},
    {"chocolate", allergen::chocolate},
    {"pollen", allergen::pollen},
    {"cats", allergen::cats}
};

}  // namespace

allergy_test::allergy_test(unsigned score) : score_(score) {}

bool allergy_test::is_allergic_to(const std::string& item) const
{
    for (const allergen_entry& entry : known_allergens) {
        if (item == entry.name) {
            return is_allergic_to(entry.item);
        }
    }

    return false;
}

bool allergy_test::is_allergic_to(allergen item) const
{
    const unsigned item_value = static_cast<unsigned>(item);
    return (score_ & item_value) != 0;
}

std::unordered_set<std::string> allergy_test::get_allergies() const
{
    std::unordered_set<std::string> result;

    for (const allergen_entry& entry : known_allergens) {
        if (is_allergic_to(entry.item)) {
            result.emplace(entry.name);
        }
    }

    return result;
}

}  // namespace allergies
