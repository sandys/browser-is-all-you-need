The supplied files need the `space_age` class and planet-specific age conversion methods. The age is stored in seconds, and each method converts it to Earth years before accounting for the planet’s orbital period.

space_age.h
```
#if !defined(SPACE_AGE_H)
#define SPACE_AGE_H

namespace space_age {

class space_age {
public:
    explicit space_age(long long seconds);

    long long seconds() const;

    double on_mercury() const;
    double on_venus() const;
    double on_earth() const;
    double on_mars() const;
    double on_jupiter() const;
    double on_saturn() const;
    double on_uranus() const;
    double on_neptune() const;

    long long seconds_;
};

}  // namespace space_age

#endif  // SPACE_AGE_H

space_age.cpp
```
#include "space_age.h"

namespace space_age {

namespace {
constexpr double seconds_per_earth_year = 31557600.0;

constexpr double mercury_years = 0.2408467;
constexpr double venus_years = 0.61519726;
constexpr double earth_years = 1.0;
constexpr double mars_years = 1.8808158;
constexpr double jupiter_years = 11.862615;
constexpr double saturn_years = 29.447498;
constexpr double uranus_years = 84.016846;
constexpr double neptune_years = 164.79132;

double age_in_years(long long seconds, double orbital_period) {
    return static_cast<double>(seconds) /
           (seconds_per_earth_year * orbital_period);
}
}  // namespace

space_age::space_age(long long seconds) : seconds_(seconds) {}

long long space_age::seconds() const {
    return seconds_;
}

double space_age::on_mercury() const {
    return age_in_years(seconds_, mercury_years);
}

double space_age::on_venus() const {
    return age_in_years(seconds_, venus_years);
}

double space_age::on_earth() const {
    return age_in_years(seconds_, earth_years);
}

double space_age::on_mars() const {
    return age_in_years(seconds_, mars_years);
}

double space_age::on_jupiter() const {
    return age_in_years(seconds_, jupiter_years);
}

double space_age::on_saturn() const {
    return age_in_years(seconds_, saturn_years);
}

double space_age::on_uranus() const {
    return age_in_years(seconds_, uranus_years);
}

double space_age::on_neptune() const {
    return age_in_years(seconds_, neptune_years);
}

}  // namespace space_age
