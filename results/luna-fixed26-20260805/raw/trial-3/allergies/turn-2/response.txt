The compilation errors occur because the tests pass allergen names as strings, while `is_allergic_to` currently accepts an `Allergen` enum. The method should accept a `std::string` and map recognized names to their corresponding bit values. Unknown names return `false`.

The CMake deprecation warning is unrelated to these errors and does not need to be changed in the requested files.

enum class Allergen : unsigned {

    pollen = 64,

class allergy_test {

    bool is_allergic_to(const std::string& allergen) const;
    std::unordered_set<Allergen> get_allergies() const;

#include "allergies.h"

namespace allergies {

allergy_test::allergy_test(unsigned score) : score_(score) {}

bool allergy_test::is_allergic_to(const std::string& allergen) const
{
    if (allergen == "eggs") {
        return (score_ & static_cast<unsigned>(Allergen::eggs)) != 0;
    }

    if (allergen == "peanuts") {
        return (score_ & static_cast<unsigned>(Allergen::peanuts)) != 0;
    }

    if (allergen == "shellfish") {
        return (score_ & static_cast<unsigned>(Allergen::shellfish)) != 0;
    }

    if (allergen == "strawberries") {
        return (score_ & static_cast<unsigned>(Allergen::strawberries)) != 0;
    }

    if (allergen == "tomatoes") {
        return (score_ & static_cast<unsigned>(Allergen::tomatoes)) != 0;
    }

    if (allergen == "chocolate") {
        return (score_ & static_cast<unsigned>(Allergen::chocolate)) != 0;
    }

    if (allergen == "pollen") {
        return (score_ & static_cast<unsigned>(Allergen::pollen)) != 0;
    }

    if (allergen == "cats") {
        return (score_ & static_cast<unsigned>(Allergen::cats)) != 0;
    }

    return false;
}

std::unordered_set<Allergen> allergy_test::get_allergies() const
{
    const Allergen allergens[] = {
        Allergen::eggs,
        Allergen::peanuts,
        Allergen::shellfish,
        Allergen::strawberries,
        Allergen::tomatoes,
        Allergen::chocolate,
        Allergen::pollen,
        Allergen::cats
    };

    std::unordered_set<Allergen> result;

    for (const auto allergen : allergens) {
        if ((score_ & static_cast<unsigned>(allergen)) != 0) {
            result.emplace(allergen);
        }
    }

    return result;
}

}  // namespace allergies
