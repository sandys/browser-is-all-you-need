The compilation errors occur because the tests call `is_allergic_to` with allergen names such as `"eggs"`, while the current function only accepts an `allergy` enum value.

The code below adds the required string-based overload and maps recognized names to their corresponding allergy values. Unknown names return `false`, and scores containing unlisted allergy bits remain ignored.

allergies.h
```
#if !defined(ALLERGIES_H)
#define ALLERGIES_H

#include <string>
#include <unordered_set>

namespace allergies {

enum class allergy : unsigned {
    eggs = 1,
    peanuts = 2,
    shellfish = 4,
    strawberries = 8,
    tomatoes = 16,
    chocolate = 32,
    pollen = 64,
    cats = 128
};

class allergy_test {
public:
    explicit allergy_test(unsigned score);

    bool is_allergic_to(const std::string& allergen) const;
    bool is_allergic_to(allergy allergen) const;

    std::unordered_set<allergy> get_allergies() const;

private:
    unsigned score_;
};

}  // namespace allergies

#endif  // ALLERGIES_H
```

allergies.cpp
```
#include "allergies.h"

namespace allergies {

namespace {

bool allergy_from_name(const std::string& name, allergy& result)
{
    if (name == "eggs") {
        result = allergy::eggs;
    } else if (name == "peanuts") {
        result = allergy::peanuts;
    } else if (name == "shellfish") {
        result = allergy::shellfish;
    } else if (name == "strawberries") {
        result = allergy::strawberries;
    } else if (name == "tomatoes") {
        result = allergy::tomatoes;
    } else if (name == "chocolate") {
        result = allergy::chocolate;
    } else if (name == "pollen") {
        result = allergy::pollen;
    } else if (name == "cats") {
        result = allergy::cats;
    } else {
        return false;
    }

    return true;
}

}  // namespace

allergy_test::allergy_test(unsigned score) : score_(score) {}

bool allergy_test::is_allergic_to(const std::string& allergen) const
{
    allergy allergen_value{};

    if (!allergy_from_name(allergen, allergen_value)) {
        return false;
    }

    return is_allergic_to(allergen_value);
}

bool allergy_test::is_allergic_to(allergy allergen) const
{
    const auto allergen_value = static_cast<unsigned>(allergen);
    return (score_ & allergen_value) != 0;
}

std::unordered_set<allergy> allergy_test::get_allergies() const
{
    std::unordered_set<allergy> result;

    for (unsigned value = static_cast<unsigned>(allergy::eggs);
         value <= static_cast<unsigned>(allergy::cats);
         value <<= 1) {
        const auto allergen = static_cast<allergy>(value);

        if (is_allergic_to(allergen)) {
            result.emplace(allergen);
        }
    }

    return result;
}

}  // namespace allergies
```
