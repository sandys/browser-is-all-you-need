The CMake warnings are unrelated to the compilation failure. The tests expect the older `meetup::scheduler` API, including Boost Gregorian dates and methods such as `monteenth()` and `tuesteenth()`. The header and implementation need to provide that API while retaining the existing `Meetup` interface.

#include "meetup.h"

#include <iomanip>
#include <sstream>
#include <string>

namespace meetup {
namespace {

int month_number(boost::gregorian::greg_month month)
{
    return month.as_number();
}

boost::gregorian::date make_date(
    int year,
    int month,
    int day)
{
    return boost::gregorian::date(
        year,
        static_cast<unsigned short>(month),
        static_cast<unsigned short>(day));
}

int days_in_month(int year, int month)
{
    return static_cast<int>(
        boost::gregorian::gregorian_calendar::end_of_month_day(
            year,
            static_cast<unsigned short>(month)));
}

int weekday_number(const boost::gregorian::date& date)
{
    return static_cast<int>(date.day_of_week());
}

std::string format_date(int year, int month, int day)
{
    std::ostringstream result;
    result << std::setfill('0')
           << std::setw(4) << year << '-'
           << std::setw(2) << month << '-'
           << std::setw(2) << day;
    return result.str();
}

}  // namespace

scheduler::scheduler(boost::gregorian::greg_month month, int year)
    : month_(month), year_(year)
{
}

boost::gregorian::date scheduler::find_date(
    Weekday weekday,
    Schedule schedule) const
{
    const int month = month_number(month_);
    const int target_weekday = static_cast<int>(weekday);
    const int last_day = days_in_month(year_, month);

    if (schedule == Schedule::teenth) {
        for (int day = 13; day <= 19; ++day) {
            const auto date = make_date(year_, month, day);
            if (weekday_number(date) == target_weekday) {
                return date;
            }
        }
    }

    if (schedule == Schedule::last) {
        for (int day = last_day; day >= 1; --day) {
            const auto date = make_date(year_, month, day);
            if (weekday_number(date) == target_weekday) {
                return date;
            }
        }
    }

    int occurrence = 0;

    switch (schedule) {
    case Schedule::first:
        occurrence = 1;
        break;
    case Schedule::second:
        occurrence = 2;
        break;
    case Schedule::third:
        occurrence = 3;
        break;
    case Schedule::fourth:
        occurrence = 4;
        break;
    case Schedule::last:
    case Schedule::teenth:
        break;
    }

    int matching_days = 0;

    for (int day = 1; day <= last_day; ++day) {
        const auto date = make_date(year_, month, day);

        if (weekday_number(date) == target_weekday) {
            ++matching_days;

            if (matching_days == occurrence) {
                return date;
            }
        }
    }

    return boost::gregorian::date();
}

std::string Meetup::day(Weekday weekday, Schedule schedule) const
{
    scheduler calendar(
        boost::gregorian::greg_month(
            static_cast<unsigned short>(month_)),
        year_);

    boost::gregorian::date result;

    switch (weekday) {
    case Weekday::Sunday:
        switch (schedule) {
        case Schedule::first: result = calendar.sunfirst(); break;
        case Schedule::second: result = calendar.sunsecond(); break;
        case Schedule::third: result = calendar.sunthird(); break;
        case Schedule::fourth: result = calendar.sunfourth(); break;
        case Schedule::last: result = calendar.sunlast(); break;
        case Schedule::teenth: result = calendar.sunteenth(); break;
        }
        break;

    case Weekday::Monday:
        switch (schedule) {
        case Schedule::first: result = calendar.monfirst(); break;
        case Schedule::second: result = calendar.monsecond(); break;
        case Schedule::third: result = calendar.monthird(); break;
        case Schedule::fourth: result = calendar.monfourth(); break;
        case Schedule::last: result = calendar.monlast(); break;
        case Schedule::teenth: result = calendar.monteenth(); break;
        }
        break;

    case Weekday::Tuesday:
        switch (schedule) {
        case Schedule::first: result = calendar.tuefirst(); break;
        case Schedule::second: result = calendar.tuesecond(); break;
        case Schedule::third: result = calendar.tuethird(); break;
        case Schedule::fourth: result = calendar.tuefourth(); break;
        case Schedule::last: result = calendar.tuelast(); break;
        case Schedule::teenth: result = calendar.tuesteenth(); break;
        }
        break;

    case Weekday::Wednesday:
        switch (schedule) {
        case Schedule::first: result = calendar.wedfirst(); break;
        case Schedule::second: result = calendar.wedsecond(); break;
        case Schedule::third: result = calendar.wedthird(); break;
        case Schedule::fourth: result = calendar.wedfourth(); break;
        case Schedule::last: result = calendar.wedlast(); break;
        case Schedule::teenth: result = calendar.wednesteenth(); break;
        }
        break;

    case Weekday::Thursday:
        switch (schedule) {
        case Schedule::first: result = calendar.thurfirst(); break;
        case Schedule::second: result = calendar.thursecond(); break;
        case Schedule::third: result = calendar.thurthird(); break;
        case Schedule::fourth: result = calendar.thurfourth(); break;
        case Schedule::last: result = calendar.thurlast(); break;
        case Schedule::teenth: result = calendar.thursteenth(); break;
        }
        break;

    case Weekday::Friday:
        switch (schedule) {
        case Schedule::first: result = calendar.frifirst(); break;
        case Schedule::second: result = calendar.frisecond(); break;
        case Schedule::third: result = calendar.frithird(); break;
        case Schedule::fourth: result = calendar.frifourth(); break;
        case Schedule::last: result = calendar.frilast(); break;
        case Schedule::teenth: result = calendar.friteenth(); break;
        }
        break;

    case Weekday::Saturday:
        switch (schedule) {
        case Schedule::first: result = calendar.satfirst(); break;
        case Schedule::second: result = calendar.satsecond(); break;
        case Schedule::third: result = calendar.satthird(); break;
        case Schedule::fourth: result = calendar.satfourth(); break;
        case Schedule::last: result = calendar.satlast(); break;
        case Schedule::teenth: result = calendar.saturteenth(); break;
        }
        break;
    }

    if (result.is_not_a_date()) {
        return {};
    }

    return format_date(
        static_cast<int>(result.year()),
        static_cast<int>(result.month()),
        static_cast<int>(result.day()));
}
