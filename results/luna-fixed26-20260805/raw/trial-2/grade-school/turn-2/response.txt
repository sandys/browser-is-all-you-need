#include "grade_school.h"

#include <algorithm>

namespace grade_school {

void school::add(const std::string& student_name, int grade_number)
{
    for (const auto& [current_grade, students] : students_) {
        if (std::find(students.begin(), students.end(), student_name) != students.end()) {
            return;
        }
    }

    students_[grade_number].push_back(student_name);
    std::sort(students_[grade_number].begin(), students_[grade_number].end());
}

std::vector<std::string> school::grade(int grade_number) const
{
    const auto found_grade = students_.find(grade_number);

    if (found_grade == students_.end()) {
        return {};
    }

    return found_grade->second;
}

std::map<int, std::vector<std::string>> school::roster() const
{
    return students_;
}

}  // namespace grade_school
