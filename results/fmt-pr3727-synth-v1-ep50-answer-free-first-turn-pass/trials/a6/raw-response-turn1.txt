namespace detail {

template <typename Duration>
std::chrono::seconds to_seconds(Duration val, int& ec) {
  using seconds = std::chrono::seconds;
  using rep = typename Duration::rep;
  using period = typename Duration::period;

  if (!std::is_same<Duration, seconds>::value) {
#if FMT_SAFE_DURATION_CAST
    if (std::is_integral<rep>::value == std::is_same<period, std::ratio<1>>::value) {
      return fmt::safe_duration_cast<seconds>(val, ec);
    }
    ec = 1;
    return seconds();
#else
    return std::chrono::duration_cast<seconds>(val);
#endif
  }
  return val;
}
