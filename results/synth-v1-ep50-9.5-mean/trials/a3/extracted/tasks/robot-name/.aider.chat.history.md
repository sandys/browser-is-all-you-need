#include <algorithm>
#include <array>
#include <cctype>
#include <random>

namespace robot_name {
namespace {

std::mt19937 engine{std::random_device{}()};
std::uniform_int_distribution<int> letters{0, 25};
std::uniform_int_distribution<int> digits{0, 9};

std::string make_name() {
    std::string result;
    result.push_back(static_cast<char>('A' + letters(engine)));
    result.push_back(static_cast<char>('A' + letters(engine)));
    result.push_back(static_cast<char>('0' + digits(engine)));
    result.push_back(static_cast<char>('0' + digits(engine)));
    result.push_back(static_cast<char>('0' + digits(engine)));
    return result;
}

}  // namespace

robot::robot() {
    do {
        name_ = make_name();
    } while (used_names_.find(name_) != used_names_.end());
    used_names_.insert(name_);
}

std::string const& robot::name() const {
    return name_;
}

void robot::reset() {
    do {
        name_ = make_name();
    } while (used_names_.find(name_) != used_names_.end());
    used_names_.insert(name_);
}

std::set<std::string> robot::used_names_{};
}  // namespace robot_name
