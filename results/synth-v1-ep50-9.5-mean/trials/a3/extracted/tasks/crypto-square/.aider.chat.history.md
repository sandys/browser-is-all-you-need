#### # Instructions  
####   
#### Implement the classic method for composing secret messages called a square code.  
####   
#### Given an English text, output the encoded version of that text.  
####   
#### First, the input is normalized: the spaces and punctuation are removed from the English text and the message is down-cased.  
####   
#### Then, the normalized characters are broken into rows.  
#### These rows can be regarded as forming a rectangle when printed with intervening newlines.  
####   
#### For example, the sentence  
####   
#### ```text  
#### "If man was meant to stay on the ground, god would have given us roots."  
#### ```  
####   
#### is normalized to:  
####   
#### ```text  
#### "ifmanwasmeanttostayonthegroundgodwouldhavegivenusroots"  
#### ```  
####   
#### The plaintext should be organized into a rectangle as square as possible.  
#### The size of the rectangle should be decided by the length of the message.  
####   
#### If `c` is the number of columns and `r` is the number of rows, then for the rectangle `r` x `c` find the smallest possible integer `c` such that:  
####   
#### - `r * c >= length of message`,  
#### - and `c >= r`,  
#### - and `c - r <= 1`.  
####   
#### Our normalized text is 54 characters long, dictating a rectangle with `c = 8` and `r = 7`:  
####   
#### ```text  
#### "ifmanwas"  
#### "meanttos"  
#### "tayonthe"  
#### "groundgo"  
#### "dwouldha"  
#### "vegivenu"  
#### "sroots  "  
#### ```  
####   
#### The coded message is obtained by reading down the columns going left to right.  
####   
#### The message above is coded as:  
####   
#### ```text  
#### "imtgdvsfearwermayoogoanouuiontnnlvtwttddesaohghnsseoau"  
#### ```  
####   
#### Output the encoded text in chunks that fill perfect rectangles `(r X c)`, with `c` chunks of `r` length, separated by spaces.  
#### For phrases that are `n` characters short of the perfect rectangle, pad each of the last `n` chunks with a single trailing space.  
####   
#### ```text  
#### "imtgdvs fearwer mayoogo anouuio ntnnlvt wttddes aohghn  sseoau "  
#### ```  
####   
#### Notice that were we to stack these, we could visually decode the ciphertext back in to the original message:  
####   
#### ```text  
#### "imtgdvs"  
#### "fearwer"  
#### "mayoogo"  
#### "anouuio"  
#### "ntnnlvt"  
#### "wttddes"  
#### "aohghn "  
#### "sseoau "  
#### ```

#include "crypto_square.h"

#include <cctype>

namespace crypto_square {

cipher::cipher(std::string const& text) {
    normalize(text);
    calculate_dimensions();
}

std::string cipher::normalize_plain_text() const {
    return normalized_;
}

std::size_t cipher::size() const {
    return columns_;
}

std::vector<std::string> cipher::plain_text_segments() const {
    std::vector<std::string> segments;
    for (std::size_t row = 0; row < rows_; ++row) {
        segments.push_back(normalized_.substr(row * columns_, columns_));
    }
    return segments;
}

std::string cipher::cipher_text() const {
    std::string result;
    result.reserve(normalized_.size());
    for (std::size_t column = 0; column < columns_; ++column) {
        for (std::size_t row = 0; row < rows_; ++row) {
            const std::size_t position = row * columns_ + column;
            if (position < normalized_.size()) {
                result.push_back(normalized_[position]);
            }
        }
    }
    return result;
}

std::string cipher::normalized_cipher_text() const {
    std::string result = cipher_text();
    if (result.empty()) {
        return result;
    }
    std::vector<std::string> segments;
    for (std::size_t column = 0; column < columns_; ++column) {
        segments.push_back(result.substr(column * rows_, rows_));
    }
    std::string normalized_result;
    for (std::string& segment : segments) {
        if (segment.size() < rows_) {
            segment.resize(rows_, ' ');
        }
        normalized_result += segment + ' ';
    }
    return normalized_result;
}

void cipher::normalize(std::string const& text) {
    for (char character : text) {
        if (std::isalnum(character) != 0) {
            normalized_.push_back(static_cast<char>(std::tolower(character)));
        }
    }
}

void cipher::calculate_dimensions() {
    if (normalized_.empty()) {
        columns_ = 0;
        rows_ = 0;
        return;
    }
    for (std::size_t column = 1; column <= normalized_.size(); ++column) {
        std::size_t row = (normalized_.size() + column - 1) / column;
        if (column >= row && column - row <= 1) {
            columns_ = column;
            rows_ = row;
            return;
        }
    }
}

}  // namespace crypto_square
