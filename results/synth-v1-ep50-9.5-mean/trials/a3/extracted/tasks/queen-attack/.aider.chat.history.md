#include "queen_attack.h"

#include <cstdlib>
#include <sstream>
#include <stdexcept>

namespace queen_attack {
namespace {

bool valid(const std::pair<int, int>& position) {
    return position.first >= 0 && position.first < 8 &&
           position.second >= 0 && position.second < 8;
}

}  // namespace

chess_board::chess_board(const std::pair<int, int>& white,
                         const std::pair<int, int>& black)
    : white_(white), black_(black) {
    if (!valid(white_) || !valid(black_)) {
        throw std::domain_error("invalid position");
    }
    if (white_ == black_) {
        throw std::domain_error("positions must be distinct");
    }
}

chess_board::chess_board()
    : chess_board({0, 3}, {7, 3}) {}

std::pair<int, int> chess_board::white() const {
    return white_;
}

std::pair<int, int> chess_board::black() const {
    return black_;
}

chess_board::operator std::string() const {
    std::ostringstream output;
    for (int row = 0; row < 8; ++row) {
        for (int col = 0; col < 8; ++col) {
            if (white_ == std::make_pair(row, col)) {
                output << 'W';
            } else if (black_ == std::make_pair(row, col)) {
                output << 'B';
            } else {
                output << '_';
            }
        }
        output << '\n';
    }
    return output.str();
}

bool chess_board::can_attack() const {
    const int row_delta =
        std::abs(white_.first - black_.first);
    const int column_delta =
        std::abs(white_.second - black_.second);
    return row_delta == 0 || column_delta == 0 ||
           row_delta == column_delta;
}

}  // namespace queen_attack
