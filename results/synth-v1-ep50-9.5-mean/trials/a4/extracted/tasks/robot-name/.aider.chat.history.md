#include "robot_name.h"

namespace robot_name {

std::set<std::string> robot::assigned_names_;
std::mt19937 robot::engine_(std::random_device{}());
std::uniform_int_distribution<int> robot::letters_(65, 90);
std::uniform_int_distribution<int> robot::digits_(48, 57);

robot::robot() {
    reset();
}

std::string const& robot::name() const {
    return name_;
}

void robot::reset() {
    std::string candidate;
    do {
        candidate.push_back(static_cast<char>(letters_(engine_)));
        candidate.push_back(static_cast<char>(letters_(engine_)));
        candidate.push_back(static_cast<char>(digits_(engine_)));
        candidate.push_back(static_cast<char>(digits_(engine_)));
        candidate.push_back(static_cast<char>(digits_(engine_)));
    } while (assigned_names_.find(candidate) != assigned_names_.end());
    assigned_names_.insert(candidate);
    name_ = candidate;
}

}  // namespace robot_name
