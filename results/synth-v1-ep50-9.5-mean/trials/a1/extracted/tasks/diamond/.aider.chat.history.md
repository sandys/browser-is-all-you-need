#### # Instructions  
####   
#### The diamond kata takes as its input a letter, and outputs it in a diamond shape.  
#### Given a letter, it prints a diamond starting with 'A', with the supplied letter at the widest point.  
####   
#### ## Requirements  
####   
#### - The first row contains one 'A'.  
#### - The last row contains one 'A'.  
#### - All rows, except the first and last, have exactly two identical letters.  
#### - All rows have as many trailing spaces as leading spaces. (This might be 0).  
#### - The diamond is horizontally symmetric.  
#### - The diamond is vertically symmetric.  
#### - The diamond has a square shape (width equals height).  
#### - The letters form a diamond shape.  
#### - The top half has the letters in ascending order.  
#### - The bottom half has the letters in descending order.  
#### - The four corners (containing the spaces) are triangles.  
####   
#### ## Examples  
####   
#### In the following examples, spaces are indicated by `·` characters.  
####   
#### Diamond for letter 'A':  
####   
#### ```text  
#### A  
#### ```  
####   
#### Diamond for letter 'C':  
####   
#### ```text  
#### ··A··  
#### ·B·B·  
#### C···C  
#### ·B·B·  
#### ··A··  
#### ```  
####   
#### Diamond for letter 'E':  
####   
#### ```text  
#### ····A····  
#### ···B·B···  
#### ··C···C··  
#### ·D·····D·  
#### E·······E  
#### ·D·····D·  
#### ··C···C··  
#### ···B·B···  
#### ····A····  
#### ```

#include "diamond.h"

#include <cstddef>
#include <string>
#include <vector>

namespace diamond {

std::vector<std::string> rows(char middle_letter) {
    const std::size_t maximum_level =
        static_cast<std::size_t>(middle_letter - 'A');
    const std::size_t width = 2 * maximum_level + 1;
    std::vector<std::string> result;

    const auto append_level = [&](std::size_t level) {
        std::string row(width, ' ');
        const std::size_t outer = maximum_level - level;
        const std::size_t inner = level;
        const char letter = static_cast<char>('A' + level);
        const char middle = middle_letter;

        row[outer] = letter;
        row[width - 1 - outer] = letter;
        if (level != 0) {
            row[maximum_level + 1 + inner] = middle;
            row[maximum_level + inner] = middle;
        }
        result.push_back(row);
    };

    for (std::size_t level = 0; level <= maximum_level; ++level) {
        append_level(level);
    }
    for (std::size_t level = maximum_level; level-- > 0;) {
        append_level(level);
    }
    return result;
}

}  // namespace diamond
