namespace meetup {

scheduler::scheduler(boost::gregorian::date::month_type month,
                     boost::gregorian::date::year_type year)
    : month_(month), year_(year) {}

boost::gregorian::date scheduler::nth(unsigned short week,
                                      boost::gregorian::weekday weekday) const {
    boost::gregorian::nth_day_of_the_week_in_month nth(weekday, week);
    nth.set_month(month_);
    nth.set_year(year_);
    return nth.get_date();
}

boost::gregorian::date scheduler::last(boost::gregorian::weekday weekday) const {
    boost::gregorian::last_day_of_the_week_in_month last(weekday, month_, year_);
    return last.get_date();
}

boost::gregorian::date scheduler::teenth(boost::gregorian::weekday weekday) const {
    boost::gregorian::date start(year_, month_, 13);
    boost::gregorian::date end(year_, month_, 19);
    for (boost::gregorian::date d = start; d <= end; ++d) {
        if (d.day_of_week() == weekday) {
            return d;
        }
    }
    throw std::out_of_range("teenth day not found");
}
