#include "robot_name.h"

#include <random>

namespace robot_name {
namespace {

std::mt19937& generator() {
    static std::mt19937 gen{std::random_device{}()};
    return gen;
}

}  // namespace

std::unordered_set<std::string> robot::assigned_names_;
std::mt19937 robot::generator_;
std::uniform_int_distribution<int> robot::letters_(0, 25);
std::uniform_int_distribution<int> robot::digits_(0, 9);

robot::robot() {
    reset();
}

std::string const& robot::name() const {
    return name_;
}

void robot::reset() {
    std::uniform_int_distribution<int> letter_distro(65, 90);
    std::uniform_int_distribution<int> digit_distro(48, 57);

    std::string new_name;
    do {
        new_name.push_back(static_cast<char>(letter_distro(generator())));
        new_name.push_back(static_cast<char>(letter_distro(generator())));
        new_name.push_back(static_cast<char>(digit_distro(generator())));
        new_name.push_back(static_cast<char>(digit_distro(generator())));
        new_name.push_back(static_cast<char>(digit_distro(generator())));
    } while (assigned_names_.find(new_name) != assigned_names_.end());

    assigned_names_.insert(new_name);
    name_ = std::move(new_name);
}

}  // namespace robot_name
