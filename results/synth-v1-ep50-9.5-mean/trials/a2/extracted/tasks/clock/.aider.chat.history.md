#include "clock.h"

#include <iomanip>
#include <sstream>

namespace date_independent {

clock::clock(long long minutes) : minutes_(minutes) {}

clock clock::at(int hour, int minute) {
    return clock(
        static_cast<long long>(hour) * 60LL + minute);
}

clock& clock::plus(int minutes) {
    minutes_ += minutes;
    minutes_ %= 1440LL;
    if (minutes_ < 0) {
        minutes_ += 1440LL;
    }
    return *this;
}

clock& clock::minus(int minutes) {
    minutes_ -= minutes;
    minutes_ %= 1440LL;
    if (minutes_ < 0) {
        minutes_ += 1440LL;
    }
    return *this;
}

clock::operator std::string() const {
    std::ostringstream output;
    output << std::setfill('0') << std::setw(2) << minutes_ / 60 << ':' 
           << std::setfill('0') << std::setw(2) << minutes_ % 60;
    return output.str();
}

bool clock::operator==(const clock& rhs) const {
    return minutes_ == rhs.minutes_;
}

bool operator!=(const clock& lhs, const clock& rhs) {
    return !(lhs == rhs);
}

}  // namespace date_independent
