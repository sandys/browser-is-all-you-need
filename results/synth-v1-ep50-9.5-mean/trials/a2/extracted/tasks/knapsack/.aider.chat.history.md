#include "knapsack.h"

namespace knapsack
{

int maximum_value(int maximum_weight, const std::vector<Item>& items) {
    if (maximum_weight <= 0 || items.empty()) {
        return 0;
    }

    std::vector<int> best(maximum_weight + 1, 0);

    for (const auto& item : items) {
        if (item.weight <= 0 || item.weight > maximum_weight) {
            continue;
        }
        for (int capacity = maximum_weight; capacity >= item.weight; --capacity) {
            const auto with_item = best[capacity - item.weight] + item.value;
            if (with_item > best[capacity]) {
                best[capacity] = with_item;
            }
        }
    }

    return best[maximum_weight];
}

} // namespace knapsack
