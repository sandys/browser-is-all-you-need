#### The kindergarten class is learning about growing plants.

#include "kindergarten_garden.h"

#include <algorithm>
#include <array>
#include <string_view>

namespace kindergarten_garden {

std::array<Plants, 4> plants(std::string_view diagram, std::string_view student) {
    static constexpr std::array<std::string_view, 12> roster{
        "Alice", "Bob", "Charlie", "David", "Eve", "Fred", "Ginny", "Harriet", "Ileana", "Joseph", "Kincaid", "Larry"
    };
    
    const auto found =
        std::find(roster.begin(), roster.end(), student);
    if (found == roster.end()) {
        return {};
    }
    const std::size_t student_index =
        static_cast<std::size_t>(found - roster.begin());
    
    const std::size_t offset = student_index * 2;
    return {
        static_cast<Plants>(diagram[offset]),
        static_cast<Plants>(diagram[offset + 1]),
        static_cast<Plants>(diagram[offset + 12]),
        static_cast<Plants>(diagram[offset + 13])
    };
}

}  // namespace kindergarten_garden
