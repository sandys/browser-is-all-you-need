namespace robot_name {

// Initialize the static set to track used names
std::set<std::string> Robot::used_names_;

Robot::Robot() : current_name_(""), has_name_(false) {
    // Robot starts without a name
}

std::string Robot::name() const {
    if (!has_name_) {
        current_name_ = generate_name();
        has_name_ = true;
    }
    return current_name_;
}

void Robot::reset() {
    has_name_ = false;
    current_name_.clear();
}

std::string Robot::generate_name() {
    std::string name;
    std::random_device rd;
    std::mt19937 gen(rd());
    std::uniform_int_distribution<> letter_distribution(0, 25);
    std::uniform_int_distribution<> digit_distribution(0, 9);

    do {
        name.clear();
        // Generate prefix (2 uppercase letters)
        name += static_cast<char>('A' + letter_distribution(gen));
        name += static_cast<char>('A' + letter_distribution(gen));
        // Generate suffix (3 digits)
        name += static_cast<char>('0' + digit_distribution(gen));
        name += static_cast<char>('0' + digit_distribution(gen));
        name += static_cast<char>('0' + digit_distribution(gen));
    } while (used_names_.find(name) != used_names_.end());

    used_names_.insert(name);
    return name;
}

}  // namespace robot_name
