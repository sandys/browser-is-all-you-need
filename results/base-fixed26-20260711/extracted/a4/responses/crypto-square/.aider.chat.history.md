#### Implement the classic method for composing secret messages called a square code.  
####   
#### Given an English text, output the encoded version of that text.  
####   
#### First, the input is normalized: the spaces and punctuation are removed from the English text and the message is down-cased.  
####   
#### Then, the normalized characters are broken into rows.  
#### These rows can be regarded as forming a rectangle when printed with intervening newlines.  
####   
#### For example, the sentence  
####   
#### ```text  
#### "If man was meant to stay on the ground, god would have given us roots."  
#### ```  
####   
#### is normalized to:  
####   
#### ```text  
#### "ifmanwasmeanttostayonthegroundgodwouldhavegivenusroots"  
#### ```  
####   
#### The plaintext should be organized into a rectangle as square as possible.  
#### The size of the rectangle should be decided by the length of the message.  
####   
#### If `c` is the number of columns and `r` is the number of rows, then for the rectangle `r` x `c` find the smallest possible integer `c` such that:  
####   
#### - `r * c >= length of message`,  
#### - and `c >= r`,  
#### - and `c - r <= 1`.  
####   
#### Our normalized text is 54 characters long, dictating a rectangle with `c = 8` and `r = 7`:

#### "meanttos"  
#### "tayonthe"  
#### "groundgo"  
#### "dwouldha"  
#### "vegivenu"  
#### "sroots  "  
#### ```  
####   
#### The coded message is obtained by reading down the columns going left to right.  
####   
#### The message above is coded as:  
####   
#### ```text  
#### "imtgdvsfearwermayoogoanouuiontnnlvtwttddesaohghnsseoau"  
#### ```

#include <cmath>

namespace crypto_square {

std::string encode(const std::string& text) {
    // 1. Normalize: remove non-alphanumeric and lowercase
    std::string normalized;
    normalized.reserve(text.size());
    for (char c : text) {
        if (std::isalnum(static_cast<unsigned char>(c))) {
            normalized += std::tolower(static_cast<unsigned char>(c));
        }
    }

    size_t len = normalized.length();

    // Handle empty input
    if (len == 0) {
        return "";
    }

    // 2. Calculate dimensions (r x c)
    // c is the smallest integer such that c >= r and c - r <= 1
    // r * c >= len
    double sqrt_len = std::sqrt(len);
    size_t c = static_cast<size_t>(std::ceil(sqrt_len));
    size_t r = (len + c - 1) / c;

    // 3. Construct matrix (fill rows)
    // We initialize with spaces to facilitate reading columns later
    std::string matrix(r * c, ' ');
    for (size_t i = 0; i < len; ++i) {
        matrix[i] = normalized[i];
    }

    // 4. Encode: read columns, then rows
    std::string encoded;
    encoded.reserve(c * (r + 1)); // Estimate size for chunks + spaces

    for (size_t col = 0; col < c; ++col) {
        for (size_t row = 0; row < r; ++row) {
            size_t index = col * r + row;
            encoded += matrix[index];
        }
        // Add space between chunks
        if (col < c - 1) {
            encoded += ' ';
        }
    }

    return encoded;
}

}  // namespace crypto_square
