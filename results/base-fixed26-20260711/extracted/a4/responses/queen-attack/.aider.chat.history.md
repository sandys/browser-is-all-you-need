#### # Instructions  
####   
#### Given the position of two queens on a chess board, indicate whether or not they are positioned so that they can attack each other.  
####   
#### In the game of chess, a queen can attack pieces which are on the same row, column, or diagonal.  
####   
#### A chessboard can be represented by an 8 by 8 array.  
####   
#### So if you are told the white queen is at `c5` (zero-indexed at column 2, row 3) and the black queen at `f2` (zero-indexed at column 5, row 6), then you know that the set-up is like so:  
####   
#### ![A chess board with two queens. Arrows emanating from the queen at c5 indicate possible directions of capture along file, rank and diagonal.](https://assets.exercism.org/images/exercises/queen-attack/queen-capture.svg)  
####   
#### You are also able to answer whether the queens can attack each other.  
#### In this case, that answer would be yes, they can, because both pieces share a diagonal.

#include "queen_attack.h"
#include <cmath>

namespace queen_attack {

Queen::Queen(int white_row, int white_col, int black_row, int black_col)
    : _white_row(white_row),
      _white_col(white_col),
      _black_row(black_row),
      _black_col(black_col) {
    if (white_row == black_row && white_col == black_col) {
        throw std::invalid_argument("Queens cannot share the same square.");
    }
    if (white_row < 0 || white_row > 7 || white_col < 0 || white_col > 7 ||
        black_row < 0 || black_row > 7 || black_col < 0 || black_col > 7) {
        throw std::invalid_argument("Queens must be placed on the board.");
    }
}

bool Queen::can_attack() const {
    // Check if queens are on the same row
    if (_white_row == _black_row) {
        return true;
    }

    // Check if queens are on the same column
    if (_white_col == _black_col) {
        return true;
    }

    // Check if queens are on the same diagonal
    // A queen attacks along diagonals if the absolute difference 
    // between rows equals the absolute difference between columns.
    int row_difference = std::abs(_white_row - _black_row);
    int col_difference = std::abs(_white_col - _black_col);

    if (row_difference == col_difference) {
        return true;
    }

    return false;
}

}  // namespace queen_attack

#include "queen_attack.h"
#include <cmath>

namespace queen_attack {

chess_board::chess_board(std::pair<int, int> white, std::pair<int, int> black)
    : _white(white),
      _black(black) {
    if (white.first == black.first && white.second == black.second) {
        throw std::invalid_argument("Queens cannot share the same square.");
    }
    if (white.first < 0 || white.first > 7 || white.second < 0 || white.second > 7 ||
        black.first < 0 || black.first > 7 || black.second < 0 || black.second > 7) {
        throw std::invalid_argument("Queens must be placed on the board.");
    }
}

bool chess_board::can_attack() const {
    // Check if queens are on the same row
    if (get_white_row() == get_black_row()) {
        return true;
    }

    // Check if queens are on the same column
    if (get_white_col() == get_black_col()) {
        return true;
    }

    // Check if queens are on the same diagonal
    // A queen attacks along diagonals if the absolute difference 
    // between rows equals the absolute difference between columns.
    int row_difference = std::abs(get_white_row() - get_black_row());
    int col_difference = std::abs(get_white_col() - get_black_col());

    if (row_difference == col_difference) {
        return true;
    }

    return false;
}

std::pair<int, int> chess_board::white() const {
    return _white;
}

std::pair<int, int> chess_board::black() const {
    return _black;
}

}  // namespace queen_attack
