#include "bank_account.h"

namespace Bankaccount {

Bankaccount::Bankaccount() : balance(0.0), closed(false) {}

void Bankaccount::close() {
    std::lock_guard<std::mutex> lock(mtx);
    closed = true;
}

bool Bankaccount::deposit(double amount) {
    std::lock_guard<std::mutex> lock(mtx);
    if (closed) {
        return false;
    }
    balance += amount;
    return true;
}

bool Bankaccount::withdraw(double amount) {
    std::lock_guard<std::mutex> lock(mtx);
    if (closed) {
        return false;
    }
    if (amount > balance) {
        return false;
    }
    balance -= amount;
    return true;
}

} // namespace Bankaccount

    std::lock_guard<std::mutex> lock(mtx);
    if (closed) {
        throw std::runtime_error("Account is closed");

    std::lock_guard<std::mutex> lock(mtx);
    closed = true;

    std::lock_guard<std::mutex> lock(mtx);
    if (closed) {
        return false;
    }
    balance += amount;
    return true;

    std::lock_guard<std::mutex> lock(mtx);
    if (closed) {
        return false;

        return false;
    }
    balance -= amount;
    return true;
}
