#include "all_your_base.h"
#include <algorithm>

namespace all_your_base {

std::vector<unsigned int> rebase(unsigned int in_base, const std::vector<unsigned int>& digits, unsigned int out_base) {
    // Validation: Bases must be at least 2
    if (in_base < 2 || out_base < 2) {
        return {};
    }

    // Handle empty input: 0 is represented as [0]
    if (digits.empty()) {
        return {0};
    }

    // Step 1: Convert input string of digits to a decimal integer
    unsigned int decimal_value = 0;
    for (unsigned int digit : digits) {
        // Validation: Digits must be valid for the input base
        if (digit >= in_base) {
            return {};
        }
        decimal_value = decimal_value * in_base + digit;
    }

    // If the number is 0, return [0] immediately
    if (decimal_value == 0) {
        return {0};
    }

    // Step 2: Convert decimal integer to the output base
    std::vector<unsigned int> result;
    while (decimal_value > 0) {
        result.push_back(decimal_value % out_base);
        decimal_value /= out_base;
    }

    // The digits were collected from least to most significant, so reverse them
    std::reverse(result.begin(), result.end());

    return result;
}

}  // namespace all_your_base
