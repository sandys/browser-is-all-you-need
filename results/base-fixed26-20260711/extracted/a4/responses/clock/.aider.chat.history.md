namespace date_independent {

Clock::Clock(int hours, int minutes) : _hours(hours), _minutes(minutes) {
    normalize();

Clock Clock::add(int minutes) const {
    return Clock(_hours, _minutes + minutes);
}

Clock Clock::subtract(int minutes) const {

}

bool Clock::operator==(const Clock& other) const {
    return _hours == other._hours && _minutes == other._minutes;
}

    std::string hours = std::to_string(clock._hours);
    std::string minutes = std::to_string(clock._minutes);

    if (clock._minutes < 10) minutes = "0" + minutes;

}

    int total_minutes = _hours * 60 + _minutes;
    total_minutes = total_minutes % (24 * 60);
    if (total_minutes < 0) {
        total_minutes += 24 * 60;
    }
    _hours = total_minutes / 60;
    _minutes = total_minutes % 60;
}

}  // namespace date_independent
