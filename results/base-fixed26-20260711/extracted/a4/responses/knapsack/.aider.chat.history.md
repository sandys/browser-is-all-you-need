#include "knapsack.h"
#include <algorithm>

namespace knapsack
{

int solve(const std::vector<Item>& items, int capacity)
{
    // dp[w] stores the maximum value achievable with weight limit w
    std::vector<int> dp(capacity + 1, 0);

    for (const auto& item : items)
    {
        // Iterate backwards to ensure we do not use the same item more than once
        for (int w = capacity; w >= item.weight; --w)
        {
            int valueWithItem = dp[w - item.weight] + item.value;
            if (valueWithItem > dp[w])
            {
                dp[w] = valueWithItem;
            }
        }
    }

    return dp[capacity];
}

} // namespace knapsack

#include "knapsack.h"
#include <algorithm>

namespace knapsack
{

int maximum_value(int capacity, const std::vector<Item>& items)
{
    // dp[w] stores the maximum value achievable with weight limit w
    std::vector<int> dp(capacity + 1, 0);

    for (const auto& item : items)
    {
        // Iterate backwards to ensure we do not use the same item more than once
        for (int w = capacity; w >= item.weight; --w)
        {
            int valueWithItem = dp[w - item.weight] + item.value;
            if (valueWithItem > dp[w])
            {
                dp[w] = valueWithItem;
            }
        }
    }

    return dp[capacity];
}

} // namespace knapsack
