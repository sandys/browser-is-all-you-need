#include "sublist.h"
#include <algorithm>

namespace sublist {

    bool is_sublist(const std::vector<int>& list_a, const std::vector<int>& list_b) {
        // An empty list is always a sublist of any other list
        if (list_a.empty()) {
            return true;
        }
        // We can only find list_a in list_b if list_a is not longer than list_b
        if (list_a.size() > list_b.size()) {
            return false;
        }

        // Use std::search to find if list_a appears in list_b
        return std::search(list_b.begin(), list_b.end(), list_a.begin(), list_a.end()) != list_b.end();
    }

    SublistType check_lists(const std::vector<int>& list_a, const std::vector<int>& list_b) {
        if (list_a == list_b) {
            return SublistType::EQUAL;
        }

        if (is_sublist(list_a, list_b)) {
            return SublistType::SUBLIST;
        }

        if (is_sublist(list_b, list_a)) {
            return SublistType::SUPERLIST;
        }

        return SublistType::UNEQUAL;
    }

} // namespace sublist
