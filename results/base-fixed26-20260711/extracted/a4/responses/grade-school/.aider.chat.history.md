#include "grade_school.h"
#include <algorithm>

namespace grade_school {

void school::add(const std::string& student, int grade) {
    auto& students = students_by_grade[grade];
    // Check if student already exists in this grade
    for (const auto& name : students) {
        if (name == student) {
            return; // Do not add duplicate
        }
    }
    students.push_back(student);
}

std::vector<std::string> school::grade(int grade_number) const {
    if (students_by_grade.find(grade_number) == students_by_grade.end()) {
        return {};
    }
    return students_by_grade.at(grade_number);
}

std::map<int, std::vector<std::string>> school::roster() const {
    std::map<int, std::vector<std::string>> sorted_roster;
    for (const auto& pair : students_by_grade) {
        std::vector<std::string> students = pair.second;
        std::sort(students.begin(), students.end());
        sorted_roster[pair.first] = students;
    }
    return sorted_roster;
}

}  // namespace grade_school
