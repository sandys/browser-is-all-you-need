namespace queen_attack {

Queen::Queen(int row, int col) 
    : row_(row), col_(col) 
{
    if (row < 0 || row > 7 || col < 0 || col > 7) {
        throw std::domain_error("Invalid position");
    }
}

bool Queen::can_attack(const Queen& other) const {
    return row_ == other.row_ ||
           col_ == other.col_ ||
           std::abs(row_ - other.row_) == std::abs(col_ - other.col_);
}

}  // namespace queen_attack

    : row_(row), col_(col)

    if (row < 0 || row > 7 || col < 0 || col > 7) {
        throw std::domain_error("Invalid position");
    }

bool Queen::can_attack(const Queen& other) const {
    return row_ == other.row_ ||
           col_ == other.col_ ||
           std::abs(row_ - other.row_) == std::abs(col_ - other.col_);
}

chess_board::chess_board(std::pair<int, int> white, std::pair<int, int> black) 
    : white_(white), black_(black) 
{
    Queen w(white.first, white.second);
    Queen b(black.first, black.second);

    if (w.row() == b.row() && w.col() == b.col()) {
        throw std::domain_error("Queens cannot occupy the same square");
    }
    if (w.row() < 0 || w.row() > 7 || w.col() < 0 || w.col() > 7) {
        throw std::domain_error("Invalid position");
    }
    if (b.row() < 0 || b.row() > 7 || b.col() < 0 || b.col() > 7) {
        throw std::domain_error("Invalid position");
    }
}
