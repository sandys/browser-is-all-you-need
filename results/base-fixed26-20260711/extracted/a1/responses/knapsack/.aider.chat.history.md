To solve this problem, I will implement the classic **0/1 Knapsack** solution using Dynamic Programming.

1.  **`knapsack.h`**: I added the necessary include for `std::vector` and declared the `solve` function.
2.  **`knapsack.cpp`**: I implemented the `solve` function. This function uses a 1D dynamic programming array where `dp[w]` stores the maximum value achievable with a weight limit of `w`. It iterates through items and capacity backwards to ensure each item is only used once (0/1 property), updating the maximum value for each possible weight.

knapsack.cpp
```
#include "knapsack.h"
#include <algorithm>

namespace knapsack
{

int solve(const std::vector<Item>& items, int capacity)
{
    // dp[w] will store the maximum value achievable with weight limit w
    std::vector<int> dp(capacity + 1, 0);

    for (const auto& item : items)
    {
        // Iterate backwards to ensure we only use each item once (0/1 Knapsack)
        for (int w = capacity; w >= item.weight; --w)
        {
            // If we take the current item, value becomes previous max at (w - weight) + current value
            // Otherwise, value stays the same
            dp[w] = std::max(dp[w], dp[w - item.weight] + item.value);
        }
    }

    return dp[capacity];
}

} // namespace knapsack

        for (int w = capacity; w >= item.weight; --w)
        {
            // If we take the current item, value becomes previous max at (w - weight) + current value

            dp[w] = std::max(dp[w], dp[w - item.weight] + item.value);

    }

}
