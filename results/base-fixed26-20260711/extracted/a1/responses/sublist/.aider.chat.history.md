#include "sublist.h"

namespace sublist {

bool contains(const std::vector<int>& main_list, const std::vector<int>& sub_list) {
    // An empty list is always contained in another list
    if (sub_list.empty()) return true;
    
    // If sub_list is larger than main_list, it cannot be contained
    if (sub_list.size() > main_list.size()) return false;

    // Iterate through main_list to find a matching sub-sequence
    for (size_t i = 0; i <= main_list.size() - sub_list.size(); ++i) {
        bool match = true;
        for (size_t j = 0; j < sub_list.size(); ++j) {
            if (main_list[i + j] != sub_list[j]) {
                match = false;
                break;
            }
        }
        if (match) return true;
    }
    return false;
}

Relation check_sublist(std::vector<int> list_one, std::vector<int> list_two) {
    // Check for equality
    if (list_one == list_two) {
        return Relation::EQUAL;
    }

    // Check if list_one is a sublist of list_two
    if (contains(list_one, list_two)) {
        return Relation::SUBLIST;
    }

    // Check if list_two is a sublist of list_one (meaning list_one is a superlist)
    if (contains(list_two, list_one)) {
        return Relation::SUPERLIST;
    }

    // If none of the above match, the lists are unequal
    return Relation::UNEQUAL;
}

} // namespace sublist

#include "sublist.h"

namespace sublist {

bool contains(const std::vector<int>& main_list, const std::vector<int>& sub_list) {
    // An empty list is always contained in another list
    if (sub_list.empty()) return true;
    
    // If sub_list is larger than main_list, it cannot be contained
    if (sub_list.size() > main_list.size()) return false;

    // Iterate through main_list to find a matching sub-sequence
    for (size_t i = 0; i <= main_list.size() - sub_list.size(); ++i) {
        bool match = true;
        for (size_t j = 0; j < sub_list.size(); ++j) {
            if (main_list[i + j] != sub_list[j]) {
                match = false;
                break;
            }
        }
        if (match) return true;
    }
    return false;
}

List_comparison sublist(const std::vector<int>& list_one, const std::vector<int>& list_two) {
    // Check for equality
    if (list_one == list_two) {
        return List_comparison::EQUAL;
    }

    // Check if list_one contains list_two (list_one is a superlist of list_two)
    if (contains(list_one, list_two)) {
        return List_comparison::SUPERLIST;
    }

    // Check if list_two contains list_one (list_two is a superlist of list_one, so list_one is a sublist)
    if (contains(list_two, list_one)) {
        return List_comparison::SUBLIST;
    }

    // If none of the above match, the lists are unequal
    return List_comparison::UNEQUAL;
}

} // namespace sublist
