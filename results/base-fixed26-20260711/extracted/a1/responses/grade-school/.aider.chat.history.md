#include "grade_school.h"
#include <algorithm>

namespace grade_school {

void school::add(std::string name, int grade) {
    if (name_to_grade_.find(name) != name_to_grade_.end()) {
        return;
    }
    name_to_grade_[name] = grade;
    roster_[grade].push_back(name);
    std::sort(roster_[grade].begin(), roster_[grade].end());
}

std::vector<std::string> school::students_in_grade(int grade) {
    if (roster_.find(grade) == roster_.end()) {
        return std::vector<std::string>();
    }
    return roster_[grade];
}

std::map<int, std::vector<std::string>> school::roster() {
    return roster_;
}

} // namespace grade_school
