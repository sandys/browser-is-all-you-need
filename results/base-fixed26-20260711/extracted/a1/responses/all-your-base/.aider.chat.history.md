#include "all_your_base.h"
#include <algorithm>
#include <vector>

namespace all_your_base {

    std::vector<int> convert(const std::vector<int>& digits, int from_base, int to_base) {
        // Validate bases
        if (from_base <= 1 || to_base <= 1) {
            return {};
        }

        // Validate digits
        if (digits.empty()) {
            return {};
        }

        // Convert to decimal
        int decimal_value = 0;
        for (int digit : digits) {
            if (digit < 0 || digit >= from_base) {
                return {};
            }
            decimal_value = decimal_value * from_base + digit;
        }

        // Handle the case where the number is 0
        if (decimal_value == 0) {
            return {0};
        }

        // Convert decimal to target base
        std::vector<int> result;
        while (decimal_value > 0) {
            result.push_back(decimal_value % to_base);
            decimal_value /= to_base;
        }

        // Digits were collected least significant first, so reverse them
        std::reverse(result.begin(), result.end());

        return result;
    }

}  // namespace all_your_base
