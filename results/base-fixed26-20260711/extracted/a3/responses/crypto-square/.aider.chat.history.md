namespace crypto_square {

std::string encode(const std::string& phrase) {
    // 1. Normalize: keep alphanumeric, convert to lowercase
    std::string normalized;
    for (char c : phrase) {
        if (std::isalnum(static_cast<unsigned char>(c))) {
            normalized += std::tolower(static_cast<unsigned char>(c));
        }
    }

    if (normalized.empty()) {
        return "";
    }

    // 2. Calculate dimensions (r = rows, c = columns)
    // c is the smallest integer such that c >= sqrt(L)
    int c = static_cast<int>(std::ceil(std::sqrt(normalized.size())));
    // r is the smallest integer such that r * c >= L
    int r = static_cast<int>(std::ceil(static_cast<double>(normalized.size()) / c));

    // 3. Read column-wise to form the encoded string
    std::string encoded;
    for (int col = 0; col < c; ++col) {
        for (int row = 0; row < r; ++row) {
            int index = row * c + col;
            if (index < normalized.size()) {
                encoded += normalized[index];
            }
        }
    }

    // 4. Chunk the encoded string into r-sized pieces, padding the last one
    std::ostringstream result;
    for (size_t i = 0; i < encoded.size(); i += static_cast<size_t>(r)) {
        std::string chunk = encoded.substr(i, static_cast<size_t>(r));
        // Pad with spaces if the chunk is shorter than r (happens at the end)
        while (chunk.size() < static_cast<size_t>(r)) {
            chunk += ' ';
        }
        if (i > 0) {
            result << ' ';
        }
        result << chunk;
    }

    return result.str();
}

}  // namespace crypto_square

    int c = static_cast<int>(std::ceil(std::sqrt(normalized.size())));
    // r is the smallest integer such that r * c >= L
    int r = static_cast<int>(std::ceil(static_cast<double>(normalized.size()) / c));

    // 3. Read column-wise to form the encoded string
    std::string encoded;
    for (int col = 0; col < c; ++col) {
        for (int row = 0; row < r; ++row) {
            int index = row * c + col;
            if (index < normalized.size()) {
                encoded += normalized[index];
            }
        }
    }

    // 4. Chunk the encoded string into r-sized pieces, padding the last one
    std::ostringstream result;
    for (size_t i = 0; i < encoded.size(); i += static_cast<size_t>(r)) {
        std::string chunk = encoded.substr(i, static_cast<size_t>(r));
        // Pad with spaces if the chunk is shorter than r (happens at the end)
        while (chunk.size() < static_cast<size_t>(r)) {
            chunk += ' ';
        }
        if (i > 0) {
            result << ' ';
        }
        result << chunk;
    }

    return result.str();
}
