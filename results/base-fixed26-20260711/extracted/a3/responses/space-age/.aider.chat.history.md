#include "space_age.h"

namespace space_age {

    const double EARTH_SECONDS_PER_YEAR = 31557600.0;
    const double MERCURY_ORBITAL_PERIOD = 0.2408467;
    const double VENUS_ORBITAL_PERIOD = 0.61519726;
    const double EARTH_ORBITAL_PERIOD = 1.0;
    const double MARS_ORBITAL_PERIOD = 1.8808158;
    const double JUPITER_ORBITAL_PERIOD = 11.862615;
    const double SATURN_ORBITAL_PERIOD = 29.447498;
    const double URANUS_ORBITAL_PERIOD = 84.016846;
    const double NEPTUNE_ORBITAL_PERIOD = 164.79132;

    double calculate_age(double seconds, double orbital_period) {
        return (seconds / EARTH_SECONDS_PER_YEAR) * orbital_period;
    }

    double on_earth(double seconds) {
        return calculate_age(seconds, EARTH_ORBITAL_PERIOD);
    }

    double on_mercury(double seconds) {
        return calculate_age(seconds, MERCURY_ORBITAL_PERIOD);
    }

}  // namespace space_age
