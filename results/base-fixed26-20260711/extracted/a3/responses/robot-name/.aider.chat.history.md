    name_ = generateName();

std::string Robot::generateName() {
    std::string name;
    do {
        // Use static random device and generator to avoid re-seeding every loop iteration
        static std::random_device rd;
        static std::mt19937 gen(rd());
        std::uniform_int_distribution<char> letterDist('A', 'Z');
        std::uniform_int_distribution<int> digitDist('0', '9');

        name = letterDist(gen);
        name += letterDist(gen);
        name += digitDist(gen);
        name += digitDist(gen);
        name += digitDist(gen);

    } while (existing_names_.find(name) != existing_names_.end());

    // Insert the new name into the set to ensure uniqueness
    std::lock_guard<std::mutex> lock(mtx_);
    existing_names_.insert(name);

    return name;
}
