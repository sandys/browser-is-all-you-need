#include "all_your_base.h"
#include <algorithm>

namespace all_your_base {

std::vector<int> convert(const std::vector<int>& digits, int from_base, int to_base) {
    // 1. Validate bases
    if (from_base < 2 || to_base < 2) {
        return {};
    }

    // 2. Validate input
    if (digits.empty()) {
        return {};
    }

    // 3. Skip leading zeros
    size_t start = 0;
    while (start < digits.size() && digits[start] == 0) {
        start++;
    }

    // 4. Handle case where number is 0
    if (start == digits.size()) {
        return {0};
    }

    // 5. Validate individual digits against source base
    for (size_t i = start; i < digits.size(); ++i) {
        if (digits[i] < 0 || digits[i] >= from_base) {
            return {};
        }
    }

    // 6. Convert to decimal (base 10)
    int value = 0;
    for (size_t i = start; i < digits.size(); ++i) {
        value = value * from_base + digits[i];
    }

    // 7. Convert decimal to target base
    if (value == 0) {
        return {0};
    }

    std::vector<int> result;
    while (value > 0) {
        result.insert(result.begin(), value % to_base);
        value /= to_base;
    }

    return result;
}

}  // namespace all_your_base
