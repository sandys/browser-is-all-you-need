namespace sublist {

ComparisonResult check_sublists(const std::vector<int>& A, const std::vector<int>& B) {
    // Helper lambda to check if container list contains needle list
    auto contains = [](const std::vector<int>& container, const std::vector<int>& needle) -> bool {
        if (needle.size() == 0) return true; // Empty list is always contained
        if (needle.size() > container.size()) return false;

        for (size_t i = 0; i <= container.size() - needle.size(); ++i) {
            bool match = true;
            for (size_t j = 0; j < needle.size(); ++j) {
                if (container[i + j] != needle[j]) {
                    match = false;
                    break;
                }
            }
            if (match) return true;
        }
        return false;
    };

    if (contains(A, B)) {
        if (A == B) {
            return ComparisonResult::EQUAL;
        }
        return ComparisonResult::SUBLIST;
    }

    if (contains(B, A)) {
        return ComparisonResult::SUPERLIST;
    }

    return ComparisonResult::UNEQUAL;
}

}  // namespace sublist

List_comparison sublist(const std::vector<int>& A, const std::vector<int>& B) {
    // Helper lambda to check if container list contains needle list
    auto contains = [](const std::vector<int>& container, const std::vector<int>& needle) -> bool {
        if (needle.size() == 0) return true; // Empty list is always contained
        if (needle.size() > container.size()) return false;

        for (size_t i = 0; i <= container.size() - needle.size(); ++i) {
            bool match = true;
            for (size_t j = 0; j < needle.size(); ++j) {
                if (container[i + j] != needle[j]) {
                    match = false;
                    break;
                }
            }
            if (match) return true;
        }
        return false;
    };

    if (contains(A, B)) {
        if (A == B) {
            return List_comparison::EQUAL;
        }
        return List_comparison::SUBLIST;
    }

    if (contains(B, A)) {
        return List_comparison::SUPERLIST;
    }

    return List_comparison::UNEQUAL;
}

}  // namespace sublist
