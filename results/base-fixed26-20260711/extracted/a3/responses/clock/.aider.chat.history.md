clock.cpp
```
#include "clock.h"

namespace date_independent {

Clock::Clock(int hours, int minutes) {
    this->minutes = hours * 60 + minutes;
    normalizeMinutes();
}

Clock Clock::add(int minutes) const {
    return Clock(0, this->minutes + minutes);
}

Clock Clock::subtract(int minutes) const {
    return Clock(0, this->minutes - minutes);
}

bool Clock::equals(const Clock& other) const {
    return this->minutes == other.minutes;
}

void Clock::normalizeMinutes() {
    minutes = minutes % 1440;
    if (minutes < 0) {
        minutes += 1440;
    }
}

}  // namespace date_independent
