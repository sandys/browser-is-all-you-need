#include "bank_account.h"

namespace Bankaccount {

void Bankaccount::open() {
    std::lock_guard<std::mutex> lock(mtx);
    balance = 0.0;
    is_open = true;
}

void Bankaccount::close() {
    std::lock_guard<std::mutex> lock(mtx);
    is_open = false;
}

void Bankaccount::deposit(double amount) {
    if (!is_open) {
        return;
    }
    std::lock_guard<std::mutex> lock(mtx);
    balance += amount;
}

bool Bankaccount::withdraw(double amount) {
    if (!is_open) {
        return false;
    }
    std::lock_guard<std::mutex> lock(mtx);
    if (balance >= amount) {
        balance -= amount;
        return true;
    }
    return false;
}

double Bankaccount::get_balance() {
    std::lock_guard<std::mutex> lock(mtx);
    return balance;
}

} // namespace Bankaccount

#include "bank_account.h"

namespace Bankaccount {

void Bankaccount::open() {
    std::lock_guard<std::mutex> lock(mtx);
    m_balance = 0.0;
    is_open = true;
}

void Bankaccount::close() {
    std::lock_guard<std::mutex> lock(mtx);
    is_open = false;
}

void Bankaccount::deposit(double amount) {
    if (!is_open) {
        return;
    }
    std::lock_guard<std::mutex> lock(mtx);
    m_balance += amount;
}

bool Bankaccount::withdraw(double amount) {
    if (!is_open) {
        return false;
    }
    std::lock_guard<std::mutex> lock(mtx);
    if (m_balance >= amount) {
        m_balance -= amount;
        return true;
    }
    return false;
}

double Bankaccount::get_balance() {
    std::lock_guard<std::mutex> lock(mtx);
    return m_balance;
}

double Bankaccount::balance() {
    if (!is_open) {
        throw std::runtime_error("Account is closed");
    }
    std::lock_guard<std::mutex> lock(mtx);
    return m_balance;
}

} // namespace Bankaccount
