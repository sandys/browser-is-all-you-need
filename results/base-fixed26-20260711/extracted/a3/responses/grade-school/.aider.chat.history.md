#include "grade_school.h"
#include <algorithm> // for std::lower_bound

namespace grade_school {

grade_school::grade_school() {}

bool grade_school::add(const std::string& student, int grade) {
    // Check if student already exists in the entire roster
    for (const auto& grade_students : _roster) {
        for (const auto& name : grade_students.second) {
            if (name == student) {
                return false; // Student already exists
            }
        }
    }

    // Insert student into the grade, maintaining alphabetical order
    auto& students = _roster[grade];
    auto it = std::lower_bound(students.begin(), students.end(), student);
    students.insert(it, student);

    return true;
}

std::vector<std::string> grade_school::grade(int grade) const {
    auto it = _roster.find(grade);
    if (it != _roster.end()) {
        return it->second; // Return the sorted list
    }
    return {}; // Return empty list if grade not found
}

std::map<int, std::vector<std::string>> grade_school::roster() const {
    return _roster;
}

}  // namespace grade_school
