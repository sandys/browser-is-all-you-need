#include "queen_attack.h"

namespace queen_attack {

bool can_attack(int white_q_col, int white_q_row, int black_q_col, int black_q_row) {
    // Queens can attack if they are in the same row or column
    if (white_q_row == black_q_row || white_q_col == black_q_col) {
        return true;
    }

    // Queens can attack if they are on the same diagonal
    // The absolute difference in rows must equal the absolute difference in columns
    if (std::abs(white_q_row - black_q_row) == std::abs(white_q_col - black_q_col)) {
        return true;
    }

    return false;
}

}  // namespace queen_attack
