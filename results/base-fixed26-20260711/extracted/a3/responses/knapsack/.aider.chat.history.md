#include "knapsack.h"
#include <algorithm>

namespace knapsack
{

int knapsack_value(const std::vector<Item>& items, int capacity)
{
    int n = items.size();
    
    // dp[i][w] represents the maximum value achievable with a capacity of w
    // using the first i items.
    std::vector<std::vector<int>> dp(n + 1, std::vector<int>(capacity + 1, 0));

    for (int i = 1; i <= n; ++i)
    {
        for (int w = 0; w <= capacity; ++w)
        {
            // If the weight of the current item exceeds the current capacity,
            // we cannot include it, so we carry forward the value from the previous state.
            if (items[i - 1].weight > w)
            {
                dp[i][w] = dp[i - 1][w];
            }
            else
            {
                // We have two choices:
                // 1. Exclude the item: value is dp[i-1][w]
                // 2. Include the item: value is item value + dp[i-1][w - item_weight]
                dp[i][w] = std::max(
                    dp[i - 1][w],
                    items[i - 1].value + dp[i - 1][w - items[i - 1].weight]
                );
            }
        }
    }

    // The bottom-right cell contains the maximum value for all items and full capacity.
    return dp[n][capacity];
}

} // namespace knapsack

#include "knapsack.h"
#include <algorithm>

namespace knapsack
{

int maximum_value(int capacity, const std::vector<Item>& items)
{
    int n = items.size();
    
    // dp[i][w] represents the maximum value achievable with a capacity of w
    // using the first i items.
    std::vector<std::vector<int>> dp(n + 1, std::vector<int>(capacity + 1, 0));

    for (int i = 1; i <= n; ++i)
    {
        for (int w = 0; w <= capacity; ++w)
        {
            // If the weight of the current item exceeds the current capacity,
            // we cannot include it, so we carry forward the value from the previous state.
            if (items[i - 1].weight > w)
            {
                dp[i][w] = dp[i - 1][w];
            }
            else
            {
                // We have two choices:
                // 1. Exclude the item: value is dp[i-1][w]
                // 2. Include the item: value is item value + dp[i-1][w - item_weight]
                dp[i][w] = std::max(
                    dp[i - 1][w],
                    items[i - 1].value + dp[i - 1][w - items[i - 1].weight]
                );
            }
        }
    }

    // The bottom-right cell contains the maximum value for all items and full capacity.
    return dp[n][capacity];
}

} // namespace knapsack
