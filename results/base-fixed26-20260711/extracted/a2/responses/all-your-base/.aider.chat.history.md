namespace all_your_base {

std::vector<int> conversion(const std::vector<int>& digits, int from_base, int to_base) {
    // Validate bases
    if (from_base < 2 || to_base < 2) {
        return {};
    }

    // Validate input digits
    if (digits.empty()) {
        return {};
    }

    for (int digit : digits) {
        if (digit < 0 || digit >= from_base) {
            return {};
        }
    }

    // Convert input digits to an integer value (intermediate base 10)
    int value = 0;
    for (int digit : digits) {
        value = value * from_base + digit;
    }

    // Handle the case where the value is 0
    if (value == 0) {
        return {0};
    }

    // Convert integer value to the desired base
    std::vector<int> result;
    while (value > 0) {
        int remainder = value % to_base;
        result.push_back(remainder);
        value /= to_base;
    }

    // The digits are currently in reverse order (least significant first)
    std::reverse(result.begin(), result.end());

    return result;
}

}  // namespace all_your_base
