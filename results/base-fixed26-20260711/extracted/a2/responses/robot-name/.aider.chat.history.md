#include "robot_name.h"
#include <algorithm>
#include <random>
#include <cctype>

namespace robot_name {

// Static member initialization
std::set<std::string> Robot::_names;

Robot::Robot() : _name("") {}

std::string Robot::name() const {
    if (_name.empty()) {
        _name = generate_name();
    }
    return _name;
}

void Robot::reset() {
    _names.erase(_name);
    _name.clear();
}

std::string Robot::generate_name() {
    static const std::string letters = "ABCDEFGHIJKLMNOPQRSTUVWXYZ";
    static const std::string digits = "0123456789";
    static std::random_device rd;
    static std::mt19937 generator(rd());

    std::string name;
    do {
        std::uniform_int_distribution<int> letter_dist(0, 25);
        std::uniform_int_distribution<int> digit_dist(0, 9);

        name = "";
        name += letters[letter_dist(generator)];
        name += letters[letter_dist(generator)];
        name += digits[digit_dist(generator)];
        name += digits[digit_dist(generator)];
        name += digits[digit_dist(generator)];
    } while (_names.find(name) != _names.end());

    _names.insert(name);
    return name;
}

}  // namespace robot_name

std::set<std::string> robot::_names;

robot::robot() : _name("") {}

std::string robot::name() const {
    if (_name.empty()) {
        _name = generate_name();
    }
    return _name;
}

void robot::reset() {
    _names.erase(_name);
    _name.clear();
}

std::string robot::generate_name() {
    static const std::string letters = "ABCDEFGHIJKLMNOPQRSTUVWXYZ";
    static const std::string digits = "0123456789";
    static std::random_device rd;
    static std::mt19937 generator(rd());

    std::string name;
    do {
        std::uniform_int_distribution<int> letter_dist(0, 25);
        std::uniform_int_distribution<int> digit_dist(0, 9);

        name = "";
        name += letters[letter_dist(generator)];
        name += letters[letter_dist(generator)];
        name += digits[digit_dist(generator)];
        name += digits[digit_dist(generator)];
        name += digits[digit_dist(generator)];
    } while (_names.find(name) != _names.end());

    _names.insert(name);
    return name;
}

}  // namespace robot_name
