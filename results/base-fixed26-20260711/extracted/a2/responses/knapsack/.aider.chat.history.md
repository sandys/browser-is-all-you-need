#include "knapsack.h"
#include <vector>
#include <algorithm>

namespace knapsack
{

int solve(const std::vector<Item>& items, int maxWeight)
{
    int n = items.size();
    int W = maxWeight;

    // Create a DP table initialized to 0.
    // dp[i][w] stores the maximum value achievable with a knapsack capacity of 'w'
    // using the first 'i' items.
    std::vector<std::vector<int>> dp(n + 1, std::vector<int>(W + 1, 0));

    for (int i = 1; i <= n; i++)
    {
        for (int w = 1; w <= W; w++)
        {
            // Check if the current item can fit into the knapsack.
            if (items[i - 1].weight <= w)
            {
                // Decide between including the item (value + remaining capacity) 
                // or excluding it.
                dp[i][w] = std::max(
                    dp[i - 1][w],
                    dp[i - 1][w - items[i - 1].weight] + items[i - 1].value
                );
            }
            else
            {
                // If the item is too heavy, we cannot include it.
                dp[i][w] = dp[i - 1][w];
            }
        }
    }

    return dp[n][W];
}

} // namespace knapsack
