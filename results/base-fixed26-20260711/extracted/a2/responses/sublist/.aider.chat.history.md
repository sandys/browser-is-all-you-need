namespace sublist {

    // Helper function to determine if 'sub' is contained within 'super'
    bool contains(const std::vector<int>& sub, const std::vector<int>& super) {
        if (sub.empty()) {
            return true; // An empty list is a sublist of any list
        }
        
        size_t i = 0;
        for (int val : super) {
            if (val == sub[i]) {
                i++;
                if (i == sub.size()) {
                    return true;
                }
            }
        }
        return false;
    }

    Relation check_sublist(std::vector<int> list_one, std::vector<int> list_two) {
        if (list_one == list_two) {
            return Relation::EQUAL;
        }

        if (contains(list_one, list_two)) {
            return Relation::SUBLIST;
        }

        if (contains(list_two, list_one)) {
            return Relation::SUPERLIST;
        }

        return Relation::NOT_EQUAL;
    }

} // namespace sublist
