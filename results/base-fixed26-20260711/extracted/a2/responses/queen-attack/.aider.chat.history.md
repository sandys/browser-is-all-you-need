#include "queen_attack.h"
#include <cstdlib>

namespace queen_attack {

Board::Board(int white_row, int white_col,
             int black_row, int black_col)
    : white_row_(white_row), white_col_(white_col),
      black_row_(black_row), black_col_(black_col)
{
    if (white_row == black_row && white_col == black_col) {
        throw std::invalid_argument("Queens cannot occupy the same square");
    }
}

bool Board::can_attack() const {
    // Check if queens are on the same row
    if (white_row_ == black_row_) {
        return true;
    }

    // Check if queens are on the same column
    if (white_col_ == black_col_) {
        return true;
    }

    // Check if queens are on the same diagonal
    // Absolute difference in rows must equal absolute difference in columns
    if (std::abs(white_row_ - black_row_) == std::abs(white_col_ - black_col_)) {
        return true;
    }

    return false;
}

} // namespace queen_attack

    : white_(white), black_(black)
{
    if (white.first == black.first && white.second == black.second) {
        throw std::domain_error("Queens cannot occupy the same square");
    }
}

std::pair<int, int> chess_board::white() const {
    return white_;
}

std::pair<int, int> chess_board::black() const {
    return black_;
}

bool chess_board::can_attack() const {
    // Check if queens are on the same row
    if (white_.first == black_.first) {
        return true;
    }

    // Check if queens are on the same column
    if (white_.second == black_.second) {
        return true;
    }

    // Check if queens are on the same diagonal
    // Absolute difference in rows must equal absolute difference in columns
    if (std::abs(white_.first - black_.first) == std::abs(white_.second - black_.second)) {
        return true;
    }

    return false;
}
