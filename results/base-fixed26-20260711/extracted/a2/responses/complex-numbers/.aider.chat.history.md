#### # Instructions  
####   
#### A complex number is a number in the form `a + b * i` where `a` and `b` are real and `i` satisfies `i^2 = -1`.  
####   
#### `a` is called the real part and `b` is called the imaginary part of `z`.  
#### The conjugate of the number `a + b * i` is the number `a - b * i`.  
#### The absolute value of a complex number `z = a + b * i` is a real number `|z| = sqrt(a^2 + b^2)`. The square of the absolute value `|z|^2` is the result of multiplication of `z` by its complex conjugate.  
####   
#### The sum/difference of two complex numbers involves adding/subtracting their real and imaginary parts separately:  
#### `(a + i * b) + (c + i * d) = (a + c) + (b + d) * i`,  
#### `(a + i * b) - (c + i * d) = (a - c) + (b - d) * i`.  
####   
#### Multiplication result is by definition  
#### `(a + i * b) * (c + i * d) = (a * c - b * d) + (b * c + a * d) * i`.  
####   
#### The reciprocal of a non-zero complex number is  
#### `1 / (a + i * b) = a/(a^2 + b^2) - b/(a^2 + b^2) * i`.  
####   
#### Dividing a complex number `a + i * b` by another `c + i * d` gives:  
#### `(a + i * b) / (c + i * d) = (a * c + b * d)/(c^2 + d^2) + (b * c - a * d)/(c^2 + d^2) * i`.  
####   
#### Raising e to a complex exponent can be expressed as `e^(a + i * b) = e^a * e^(i * b)`, the last term of which is given by Euler's formula `e^(i * b) = cos(b) + i * sin(b)`.

namespace complex_numbers {

Complex::Complex(double real, double imaginary)
    : m_real(real), m_imaginary(imaginary) {}

double Complex::real() const {
    return m_real;
}

double Complex::imaginary() const {
    return m_imaginary;
}

Complex Complex::add(const Complex& other) const {
    return Complex(m_real + other.m_real, m_imaginary + other.m_imaginary);
}

Complex Complex::subtract(const Complex& other) const {
    return Complex(m_real - other.m_real, m_imaginary - other.m_imaginary);
}

Complex Complex::multiply(const Complex& other) const {
    // (a + i*b) * (c + i*d) = (a*c - b*d) + (b*c + a*d) * i
    double newReal = m_real * other.m_real - m_imaginary * other.m_imaginary;
    double newImag = m_real * other.m_imaginary + m_imaginary * other.m_real;
    return Complex(newReal, newImag);
}

Complex Complex::divide(const Complex& other) const {
    // (a + i*b) / (c + i*d) = (a*c + b*d)/(c^2 + d^2) + (b*c - a*d)/(c^2 + d^2) * i
    double denominator = other.m_real * other.m_real + other.m_imaginary * other.m_imaginary;
    double newReal = (m_real * other.m_real + m_imaginary * other.m_imaginary) / denominator;
    double newImag = (m_imaginary * other.m_real - m_real * other.m_imaginary) / denominator;
    return Complex(newReal, newImag);
}

Complex Complex::conjugate() const {
    return Complex(m_real, -m_imaginary);
}

double Complex::abs() const {
    return std::sqrt(m_real * m_real + m_imaginary * m_imaginary);
}

Complex Complex::exp() const {
    // e^(a + i*b) = e^a * e^(i*b) = e^a * (cos(b) + i * sin(b))
    double realPart = std::exp(m_real) * std::cos(m_imaginary);
    double imagPart = std::exp(m_real) * std::sin(m_imaginary);
    return Complex(realPart, imagPart);
}

} // namespace complex_numbers

#include "complex_numbers.h"

namespace complex_numbers {

Complex::Complex(double real, double imaginary)
    : m_real(real), m_imaginary(imaginary) {}

double Complex::real() const {
    return m_real;
}

double Complex::imag() const {
    return m_imaginary;
}

Complex Complex::add(const Complex& other) const {
    return Complex(m_real + other.m_real, m_imaginary + other.m_imaginary);
}

Complex Complex::subtract(const Complex& other) const {
    return Complex(m_real - other.m_real, m_imaginary - other.m_imaginary);
}

Complex Complex::multiply(const Complex& other) const {
    // (a + i*b) * (c + i*d) = (a*c - b*d) + (b*c + a*d) * i
    double newReal = m_real * other.m_real - m_imaginary * other.m_imaginary;
    double newImag = m_real * other.m_imaginary + m_imaginary * other.m_real;
    return Complex(newReal, newImag);
}

Complex Complex::divide(const Complex& other) const {
    // (a + i*b) / (c + i*d) = (a*c + b*d)/(c^2 + d^2) + (b*c - a*d)/(c^2 + d^2) * i
    double denominator = other.m_real * other.m_real + other.m_imaginary * other.m_imaginary;
    double newReal = (m_real * other.m_real + m_imaginary * other.m_imaginary) / denominator;
    double newImag = (m_imaginary * other.m_real - m_real * other.m_imaginary) / denominator;
