#include "crypto_square.h"
#include <cmath>
#include <sstream>
#include <vector>

namespace crypto_square {

std::string encode(const std::string& text) {
    // 1. Normalize: remove non-alphanumeric and lowercase
    std::string normalized;
    normalized.reserve(text.size());
    for (char c : text) {
        if (std::isalnum(static_cast<unsigned char>(c))) {
            normalized += std::tolower(static_cast<unsigned char>(c));
        }
    }

    if (normalized.empty()) {
        return "";
    }

    // 2. Calculate dimensions
    const size_t n = normalized.size();
    const size_t s = static_cast<size_t>(std::ceil(std::sqrt(n)));
    const size_t c = s;
    const size_t r = (n + c - 1) / c;

    // 3. Read column by column
    std::vector<std::string> chunks(c);
    for (size_t col = 0; col < c; ++col) {
        std::string chunk;
        chunk.reserve(r);
        for (size_t row = 0; row < r; ++row) {
            const size_t index = row * c + col;
            if (index < n) {
                chunk += normalized[index];
            } else {
                chunk += ' '; // Padding space
            }
        }
        chunks[col] = chunk;
    }

    // 4. Join with spaces
    std::ostringstream oss;
    for (size_t i = 0; i < chunks.size(); ++i) {
        if (i > 0) {
            oss << ' ';
        }
        oss << chunks[i];
    }
    return oss.str();
}

}  // namespace crypto_square
